"banner rate"
// Modifies how quickly the banner fills
"banner speed"
// Modifies how quickly the player moves while the banner is active
"banner effect"
// Condition ONLY the banner owner gets while the effect is active

"banner ammo"
// Banners give this percentage in ammo every application (so once every second)
"banner heads"
